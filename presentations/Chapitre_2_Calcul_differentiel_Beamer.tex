% 03. Derivee / tangente.
\begin{frame}{1. Dérivée : mesurer une variation instantanée}
\begin{block}{Définition en un point intérieur $a$}
\[f'(a)=\lim_{h\to0}\frac{f(a+h)-f(a)}{h},\quad\text{si cette limite existe et est finie.}\]
La tangente au graphe de $f$ en $a$ a pour équation
\[y=f(a)+f'(a)(x-a).\]
\end{block}
\begin{exampleblock}{Exemple : $f(x)=x^2$ en $a=2$}
\[\frac{(2+h)^2-4}{h}=4+h\to4;\qquad y=4+4(x-2)=4x-4.\]
\end{exampleblock}
\smallskip
{\small\color{Corail}\textbf{Attention :} dérivable en $a$ $\Rightarrow$ continue en $a$ ; réciproque fausse ($|x|$ en 0).}
\end{frame}

% 04. Regles et composees.
\begin{frame}{2. Calculer les dérivées sans se tromper}
Pour $u,v$ dérivables et $v\ne0$ :
\[ (u+v)'=u'+v',\quad (uv)'=u'v+uv',\quad
\left(\frac uv\right)'=\frac{u'v-uv'}{v^2}. \]
\begin{block}{Composition : règle de la chaîne}
\[(g\circ u)'(x)=g'(u(x))\,u'(x).\]
\vspace{-.3cm}
\begin{tabularx}{\textwidth}{@{}XX@{}}
$(u^n)'=nu^{n-1}u'$ & $(\e^u)'=u'\e^u$\\
$(\ln u)'=u'/u$ ($u>0$) & $(\sin u)'=u'\cos u$\\
$(\cos u)'=-u'\sin u$ & $(\sqrt u)'=u'/(2\sqrt u)$ ($u>0$)
\end{tabularx}
\end{block}
\begin{exampleblock}{Application immédiate}
$f(x)=(x^2+1)\e^{-x}$ : \quad $f'(x)=(-x^2+2x-1)\e^{-x}=-(x-1)^2\e^{-x}$.
\end{exampleblock}
\end{frame}

% 05. Derivees successives.
\begin{frame}{3. Dérivées successives et interprétation}
\begin{block}{Définition}
$f''=(f')'$, $f'''=(f'')'$, puis $f^{(n)}$. La dérivée seconde mesure l'évolution de la pente.
\end{block}
\begin{exampleblock}{Calcul simple}
Pour $f(x)=\e^{2x}$ :
\[f'=2\e^{2x},\qquad f''=4\e^{2x},\qquad f^{(n)}=2^n\e^{2x}.\]
\end{exampleblock}
\begin{alertblock}{Signification physique des unités}
Si $Q(t)$ est en $\mathrm{m^3/s}$ et $t$ en heures, alors $Q'(t)$ est en $\mathrm{m^3/(s\,h)}$ : un débit instantané et sa variation n'ont pas la même unité.
\end{alertblock}
\end{frame}

% 06. Rolle / TAF.
\begin{frame}{4. Les deux théorèmes fondamentaux}
\begin{block}{Théorème de Rolle}
Si $f$ est \textbf{continue} sur $[a,b]$, \textbf{dérivable} sur $]a,b[$ et $f(a)=f(b)$, alors il existe $c\in]a,b[$ tel que $f'(c)=0$.
\end{block}
\begin{block}{Théorème des accroissements finis (TAF)}
Sous les mêmes hypothèses de continuité et dérivabilité (sans $f(a)=f(b)$) :
\[\exists c\in]a,b[,\qquad f'(c)=\frac{f(b)-f(a)}{b-a}.\]
\end{block}
\begin{exampleblock}{Exemple}
$f(x)=x^2$ sur $[1,3]$ : pente moyenne $4$ ; $2c=4$, donc $c=2$.
\end{exampleblock}
\end{frame}

% 07. Variations et extrema.
\begin{frame}{5. Variations et extrema : une méthode}
\begin{enumerate}
\item Déterminer le domaine $D_f$ et calculer $f'$.
\item Résoudre $f'(x)=0$ et repérer les points où $f'$ n'existe pas.
\item Étudier le signe de $f'$ sur chaque intervalle.
\item Dresser le tableau de variations ; comparer les valeurs aux bornes si elles existent.
\end{enumerate}
\begin{block}{Critères du premier ordre}
$f'>0$ sur un intervalle $\Rightarrow f$ strictement croissante ; $f'<0$ $\Rightarrow$ strictement décroissante. Changement $+\to-$ : maximum local ; $-\to+$ : minimum local.
\end{block}
\begin{exampleblock}{Exemple}
$f(x)=x^3-3x$, $f'(x)=3(x-1)(x+1)$ : croît sur $]-\infty,-1]$, décroît sur $[-1,1]$, croît sur $[1,+\infty[$.
\end{exampleblock}
\end{frame}

% 08. Second ordre.
\begin{frame}{6. Dérivée seconde : courbure et extrema}
\begin{block}{Sur un intervalle}
$f''\ge0$ $\Rightarrow$ $f$ \textbf{convexe} ; $f''\le0$ $\Rightarrow$ $f$ \textbf{concave}.\\
Si $f'(a)=0$ et $f''(a)>0$ : minimum local strict ; si $f''(a)<0$ : maximum local strict.
\end{block}
\begin{exampleblock}{Exemple : $f(x)=x^3-3x$}
$f''(x)=6x$. $f''(-1)=-6<0$ : maximum local $f(-1)=2$.\\
$f''(1)=6>0$ : minimum local $f(1)=-2$.\\
$f''$ change de signe en $0$ : point d'inflexion $(0,0)$.
\end{exampleblock}
\begin{alertblock}{Nuance}
$f''(a)=0$ ne permet pas, à lui seul, de conclure à un extremum ou à une inflexion.
\end{alertblock}
\end{frame}

% 09. Taylor local.
\begin{frame}{7. Développement limité : approcher localement}
\begin{block}{Taylor--Young à l'ordre 2, au voisinage de $a$}
Si $f$ est de classe $\mathcal C^2$ près de $a$, alors
\[f(a+h)=f(a)+f'(a)h+\frac{f''(a)}{2}h^2+o(h^2),\quad h\to0.\]
\end{block}
\begin{exampleblock}{Exemple au voisinage de $0$}
\[\e^x=1+x+\frac{x^2}{2}+o(x^2).\]
Donc $\e^{-0.1}\approx1-0.1+0.005=0.905$ (valeur réelle $\approx0.904837$).
\end{exampleblock}
{\small\color{Corail}\textbf{Attention :} $o(h^2)/h^2\to0$ quand $h\to0$ ; le DL est \textbf{local}.}
\end{frame}

% 10. DL usuels.
\begin{frame}{8. Développements limités usuels en $0$}
\begin{tabularx}{\textwidth}{@{}lX@{}}
\toprule
Fonction & Développement au voisinage de $0$\\
\midrule
$\e^x$ & $1+x+\frac{x^2}{2}+o(x^2)$\\[2mm]
$\ln(1+x)$ & $x-\frac{x^2}{2}+o(x^2)$ ($x>-1$)\\[2mm]
$\sin x$ & $x-\frac{x^3}{6}+o(x^3)$\\[2mm]
$\cos x$ & $1-\frac{x^2}{2}+o(x^2)$\\[2mm]
$(1+x)^\alpha$ & $1+\alpha x+\frac{\alpha(\alpha-1)}2x^2+o(x^2)$ ($1+x>0$)\\
\bottomrule
\end{tabularx}
\begin{exampleblock}{Lever une indétermination}
\[\lim_{x\to0}\frac{\ln(1+x)-x}{x^2}=-\frac12.\]
\end{exampleblock}
\end{frame}

% 11. Restes.
\begin{frame}{9. Une approximation utile et son erreur}
\begin{block}{Taylor--Lagrange : majorer l'erreur}
Si $|f^{(3)}(x)|\le M$ entre $a$ et $a+h$, alors
\[\left|f(a+h)-f(a)-f'(a)h-\frac{f''(a)}{2}h^2\right|\le\frac{M|h|^3}{6}.\]
\end{block}
\begin{exampleblock}{Exemple : prédire près d'un minimum}
$f(x)=x^3-3x$, autour de $a=1$ : $f(1)=-2$, $f'(1)=0$, $f''(1)=6$.\\
\[f(1+h)=-2+3h^2+h^3.\]
À $h=0.2$ : DL d'ordre 2 $=-1.88$ ; valeur exacte $=-1.872$.\\
Erreur $=0.008$ ; la borne avec $M=6$ vaut $0.008$.
\end{exampleblock}
\end{frame}
